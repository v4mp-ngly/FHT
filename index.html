La cedula valida al paciente y genera un token corto -- que sirve para guardar la ficha sin exponer la tabla de pacientes. create or replace function public.numed_photobooth_login(p_cedula text) returns table( paciente_id uuid, nombre text, edad_cronologica integer, edad_biologica numeric, token text ) language plpgsql security definer set search_path=public as $$ declare v_cedula text := regexp_replace(coalesce(p_cedula,''),'[^0-9]','','g'); v_patient uuid; v_token text; begin if length(v_cedula) <> 11 then raise exception 'Cedula no valida'; end if; select p.id into v_patient from public.pacientes_evaluacion p where regexp_replace(coalesce(p.cedula,''),'[^0-9]','','g')=v_cedula and exists ( select 1 from public.mediciones_edad_biologica m where m.paciente_id=p.id and m.edad_biologica is not null ) order by p.created_at desc limit 1; if v_patient is null then raise exception 'No encontramos una evaluacion lista con esa cedula'; end if; delete from public.photobooth_access_tokens as pat where pat.expires_at < now() or (pat.paciente_id = v_patient and pat.created_at < now()-interval '2 minutes'); v_token := encode(gen_random_bytes(32),'hex'); insert into public.photobooth_access_tokens(paciente_id,token,expires_at) values(v_patient,v_token,now()+interval '30 minutes'); return query select p.id, trim(concat_ws(' ',p.nombres,p.apellidos))::text, extract(year from age(current_date,p.fecha_nacimiento))::integer, m.edad_biologica, v_token from public.pacientes_evaluacion p join lateral ( select mb.edad_biologica from public.mediciones_edad_biologica mb where mb.paciente_id=p.id and mb.edad_biologica is not null order by mb.created_at desc limit 1 ) m on true where p.id=v_patient; end $$; revoke all on function public.numed_photobooth_login(text) from public; grant execute on function public.numed_photobooth_login(text) to anon,authenticated; -- Mantiene compatibilidad con el guardado existente del Photo Booth. -- Si numed_save_photobooth_card ya existe en tu proyecto, no se modifica aqui. notify pgrst,'reload schema';
